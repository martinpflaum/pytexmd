# A running notebook does not see PATH updates made by a MiKTeX installer.
# Check standard per-user and system Windows installations. Linux TeX
# distributions install their executables on PATH and need no special case.
if _pytexmd_sys.platform == 'win32':
    _pytexmd_miktex_bins = []
    for _pytexmd_root_name, _pytexmd_parts in (
        ('LOCALAPPDATA', ('Programs', 'MiKTeX', 'miktex', 'bin')),
        ('PROGRAMFILES', ('MiKTeX', 'miktex', 'bin')),
    ):
        _pytexmd_root = _pytexmd_os.environ.get(_pytexmd_root_name)
        if _pytexmd_root:
            _pytexmd_base = _PytexmdPath(_pytexmd_root).joinpath(*_pytexmd_parts)
            _pytexmd_miktex_bins.extend((_pytexmd_base / 'x64', _pytexmd_base))
    for _pytexmd_miktex_bin in _pytexmd_miktex_bins:
        if _pytexmd_miktex_bin.is_dir():
            _pytexmd_os.environ['PATH'] = (
                str(_pytexmd_miktex_bin)
                + _pytexmd_os.pathsep
                + _pytexmd_os.environ.get('PATH', '')
            )
            break

# sphinxcontrib-tikz always emits its closing </div>, even when rendering
# failed before the matching opening element was written. That malformed HTML
# can close a theme layout container and split the rest of the page into
# columns. Only run the departure callback after a successful HTML visit.
import sphinxcontrib.tikz as _pytexmd_tikz

# sphinxcontrib-tikz does not recognize Ghostscript's usual Windows command.
_pytexmd_tikz_which = _pytexmd_tikz.which


def _pytexmd_which(program):
    found = _pytexmd_tikz_which(program)
    if found is None and program in ('ghostscript', 'gs', 'gswin64'):
        return (_pytexmd_tikz_which('gswin64c') or
                _pytexmd_tikz_which('gswin32c') or
                _pytexmd_tikz_which('mgs'))
    return found


_pytexmd_tikz.which = _pytexmd_which

if not hasattr(_pytexmd_tikz, "_pytexmd_original_html_visit_tikz"):
    _pytexmd_tikz._pytexmd_original_html_visit_tikz = _pytexmd_tikz.html_visit_tikz
    _pytexmd_tikz._pytexmd_original_html_depart_tikz = _pytexmd_tikz.html_depart_tikz

_pytexmd_tikz_html_visit = _pytexmd_tikz._pytexmd_original_html_visit_tikz
_pytexmd_tikz_html_depart = _pytexmd_tikz._pytexmd_original_html_depart_tikz


def _pytexmd_html_visit_tikz(translator, node):
    body_start = len(translator.body)
    latex_engine = translator.builder.config.latex_engine
    if _pytexmd_shutil.which(latex_engine):
        _pytexmd_tikz_html_visit(translator, node)
    rendered = translator.body[body_start:]
    opened = any(
        fragment.lstrip().startswith("<div")
        for fragment in rendered
    )
    if any('src="None"' in fragment for fragment in rendered):
        del translator.body[body_start:]
        opened = False
    if not opened:
        # Keep the document useful when TeX or the configured converter is
        # unavailable. The original extension otherwise emits no content and
        # an unmatched closing div.
        source = translator.encode(node.get("tikz", "")).strip()
        translator.body.append(
            translator.starttag(node, "div", CLASS="admonition warning tikz-fallback")
        )
        translator.body.append(
            '<p class="admonition-title">TikZ diagram source</p>'
        )
        translator.body.append("<pre><code>" + source + "</code></pre>")
        opened = True
    node["_pytexmd_html_container_opened"] = opened


def _pytexmd_html_depart_tikz(translator, node):
    if node.get("_pytexmd_html_container_opened", False):
        _pytexmd_tikz_html_depart(translator, node)


_pytexmd_tikz.html_visit_tikz = _pytexmd_html_visit_tikz
_pytexmd_tikz.html_depart_tikz = _pytexmd_html_depart_tikz
